
"use client";

import { useEffect, useState } from "react";
import { supabase } from "@/lib/supabase";

type Product = {
  id: number;
  name: string;
  category: string;
  subcategory: string;
  costPrice: number;
  price: number;
  images: string[];
};

type OrderItem = {
  productId: number;
  name: string;
  quantity: number;
  costPrice: number;
  stock: number;
  price: number;
};

type Order = {
  id: string | number;
  customerName?: string;
  phone?: string;
  address?: string;
  items: OrderItem[];
  total: number;
  receipt?: string;
  status: "pending" | "confirmed" | "rejected";
  createdAt?: string;
};

const PRODUCT_KEY = "tota-products";
const ORDER_KEY = "tota-orders";

const subcategories: Record<string, string[]> = {
  "إلكترونيات": ["موبايلات", "سماعات", "إكسسوارات موبايل"],
  "أزياء": ["ملابس", "شنط", "أحذية"],
  "المنزل": ["ديكور", "أدوات منزلية"],
  "إكسسوارات": ["سلاسل", "إنسيالات", "أساور", "حلقان", "غوايش", "خواتم"],
};

const money = (value: number) =>
  new Intl.NumberFormat("ar-EG", {
    style: "currency",
    currency: "EGP",
    maximumFractionDigits: 0,
  }).format(value);

export default function AdminPage() {
  const [products, setProducts] = useState<Product[]>([]);
  const [orders, setOrders] = useState<Order[]>([]);
  const [loaded, setLoaded] = useState(false);
  const [isLoggedIn, setIsLoggedIn] = useState(false);
  const [email, setEmail] = useState("israa2002mahmoud@gmail.com");
  const [password, setPassword] = useState("");

  const [name, setName] = useState("");
  const [category, setCategory] = useState("إكسسوارات");
  const [subcategory, setSubcategory] = useState("سلاسل");
  const [costPrice, setCostPrice] = useState("");
  const [stock, setStock] = useState("0");
const [price, setPrice] = useState("");
  const [images, setImages] = useState<string[]>([]);
  const [message, setMessage] = useState("");

 
   async function handleForgotPassword() {
    if (!email.trim()) {
      setMessage("اكتبي البريد الإلكتروني أولًا.");
      return;
    }

    const { error } = await supabase.auth.resetPasswordForEmail(
      email.trim(),
      {
        redirectTo: "http://localhost:3001/reset-password",
      }
    );

    if (error) {
      setMessage(error.message);
      return;
    }

    setMessage("تم إرسال رابط تغيير كلمة المرور إلى بريدك ❤️");
  }

  async function handleLogin() {
    if (!email.trim() || !password) {
      setMessage("اكتبي الإيميل وكلمة المرور.");
      return;
    }
    const { error } = await supabase.auth.signInWithPassword({
      email: email.trim(),
      password,
    });

    if (error) {
      setMessage(error.message);
      return;
    }
    setIsLoggedIn(true);
    setPassword("");
    setMessage("تم تسجيل الدخول بنجاح ❤️");
  }
  useEffect(() => {
    async function loadData() {
      try {
        const { data: savedProducts, error: productsError } = await supabase
          .from("products")
          .select("*")
          .order("created_at", { ascending: false });

        if (productsError) throw productsError;

        setProducts((savedProducts || []).map(p => ({
          id: Number(p.id),
          name: p.name,
          category: p.category,
          subcategory: p.subcategory || "",
          costPrice: Number(p.cost_price),
          stock: Number(p.stock || 0),
          price: Number(p.price),
          images: p.images || [],
        })));

        setMessage("تم تحميل المنتجات من قاعدة البيانات ❤️");
      } catch (error) {
        console.error(error);
        setMessage("تعذر تحميل المنتجات. تأكدي من تسجيل الدخول كمديرة.");
      } finally {
        setLoaded(true);
      }
    }

    loadData();
  }, []);

  function saveProducts(updated: Product[]) {
    try {
      localStorage.setItem(PRODUCT_KEY, JSON.stringify(updated));
      setProducts(updated);
      window.dispatchEvent(new Event("tota-products-updated"));
      setMessage("تم حفظ المنتجات بنجاح ❤️");
    } catch {
      setMessage("مساحة التخزين غير كافية. قللي حجم الصور.");
    }
  }

  function saveOrders(updated: Order[]) {
    try {
      localStorage.setItem(ORDER_KEY, JSON.stringify(updated));
      setOrders(updated);
      setMessage("تم تحديث الطلب بنجاح.");
    } catch {
      setMessage("تعذر حفظ الطلبات.");
    }
  }

  function handleImages(files: FileList | null) {
    if (!files) return;

    const remaining = 5 - images.length;
    if (remaining <= 0) {
      setMessage("مسموح بحد أقصى 5 صور.");
      return;
    }

    const selected = Array.from(files).slice(0, remaining);

    if (selected.some(file => !file.type.startsWith("image/"))) {
      setMessage("اختاري صورًا فقط.");
      return;
    }

    const readers = selected.map(file => new Promise<string>((resolve, reject) => {
      const reader = new FileReader();
      reader.onload = () => resolve(reader.result as string);
      reader.onerror = () => reject();
      reader.readAsDataURL(file);
    }));

    Promise.all(readers)
      .then(newImages => {
        setImages(old => [...old, ...newImages].slice(0, 5));
        setMessage("تم تجهيز الصور.");
      })
      .catch(() => setMessage("حدث خطأ أثناء قراءة الصور."));
  }

 if (Number(price) <= 0 || Number(costPrice) < 0) {
      setMessage("راجعي الأسعار المدخلة.");
      return;
    }

    if (!Number.isInteger(Number(stock)) || Number(stock) < 0) {
      setMessage("اكتبي كمية صحيحة لا تقل عن صفر.");
      return;
    }
    const product: Product = {
      id: Date.now(),
      name: name.trim(),
      category,
      subcategory,
      costPrice: Number(costPrice),
      stock: Number(stock),
      price: Number(price),
      images,
    };

    saveProducts([product, ...products]);
    setName("");
    setCostPrice("");
    setStock("0");
    setPrice("");
    setImages([]);
  }

  function changeOrderStatus(id: string | number, status: Order["status"]) {
    const updated = orders.map(order =>
      order.id === id ? { ...order, status } : order
    );
    saveOrders(updated);
  }

  const pendingOrders = orders.filter(o => o.status === "pending");
  const confirmedOrders = orders.filter(o => o.status === "confirmed");
  const rejectedOrders = orders.filter(o => o.status === "rejected");

  const netProfit = confirmedOrders.reduce((sum, order) => {
    return sum + order.items.reduce((itemSum, item) => {
      return itemSum + (Number(item.price) - Number(item.costPrice)) * Number(item.quantity);
    }, 0);
  }, 0);

  if (!loaded) {
    return <main dir="rtl" style={pageStyle}>جاري تحميل البيانات...</main>;
  }

  if (!isLoggedIn) {
    return (
      <main dir="rtl" style={pageStyle}>
        <div style={{ maxWidth: 400, margin: "100px auto", padding: 25, background: "#24152f", borderRadius: 16 }}>
          <h1 style={{ color: "#ed70e8", textAlign: "center" }}>TOTA STORE</h1>
          <h2 style={{ textAlign: "center" }}>تسجيل دخول الإدارة</h2>

          <input
            type="email"
            placeholder="البريد الإلكتروني"
            value={email}
            onChange={(e) => setEmail(e.target.value)}
            style={{ width: "100%", padding: 12, marginBottom: 12, boxSizing: "border-box" }}
          />

          <input
            type="password"
            placeholder="كلمة المرور"
            value={password}
            onChange={(e) => setPassword(e.target.value)}
            style={{ width: "100%", padding: 12, marginBottom: 12, boxSizing: "border-box" }}
          />

          <button
            onClick={handleLogin}
            style={{ width: "100%", padding: 12, background: "#ed70e8", color: "#fff", border: 0, borderRadius: 8, cursor: "pointer" }}
          >
            تسجيل الدخول
          </button>

          <button
            onClick={handleForgotPassword}
            style={{
              width: "100%",
              padding: 12,
              marginTop: 10,
              background: "transparent",
              color: "#ed70e8",
              border: "1px solid #ed70e8",
              borderRadius: 8,
              cursor: "pointer",
            }}
          >
            نسيت كلمة المرور؟
          </button>

          <p>{message}</p>
        </div>
      </main>
    );
  }

  return (
    <main dir="rtl" style={pageStyle}>
      <div style={{ maxWidth: 1000, margin: "auto" }}>
        <h1 style={{ color: "#ed70e8" }}>TOTA STORE</h1>
        <h2>لوحة إدارة المتجر</h2>
        <p style={{ color: "#c5b8ce" }}>إدارة المنتجات والطلبات والأرباح</p>

        <h2>إحصائيات الطلبات</h2>
        <div style={statsGrid}>
          <Stat title="إجمالي الطلبات" value={orders.length} />
          <Stat title="الطلبات المؤكدة" value={confirmedOrders.length} />
          <Stat title="بانتظار المراجعة" value={pendingOrders.length} />
          <Stat title="الطلبات المرفوضة" value={rejectedOrders.length} />
          <Stat title="صافي الربح من الطلبات المؤكدة" value={money(netProfit)} />
        </div>

        <section style={sectionStyle}>
          <h2>مراجعة الطلبات وإيصالات التحويل</h2>

          {orders.length === 0 && (
            <p>لا توجد طلبات حتى الآن. ستظهر هنا الطلبات بعد ربط إتمام الشراء.</p>
          )}

          {orders.map(order => (
            <article key={order.id} style={orderStyle}>
              <h3>طلب رقم: {order.id}</h3>
              <p>العميل: {order.customerName || "غير محدد"}</p>
              <p>الهاتف: {order.phone || "غير محدد"}</p>
              <p>العنوان: {order.address || "غير محدد"}</p>
              <p>حالة الطلب: {
                order.status === "pending" ? "بانتظار مراجعة الدفع" :
                order.status === "confirmed" ? "مؤكد" : "مرفوض"
              }</p>

              <h4>المنتجات:</h4>
              {order.items.map((item, index) => (
                <p key={index}>
                  {item.name} — الكمية: {item.quantity} — سعر البيع: {money(item.price)}
                </p>
              ))}

              <p>إجمالي الطلب: {money(order.total)}</p>

              {order.receipt ? (
                <div>
                  <p>إيصال التحويل:</p>
                  <a href={order.receipt} target="_blank" rel="noreferrer">
                    <img src={order.receipt} alt="إيصال التحويل"
                      style={{ maxWidth: 220, maxHeight: 220, objectFit: "contain", borderRadius: 10 }} />
                  </a>
                </div>
              ) : (
                <p>لم يتم رفع إيصال.</p>
              )}

              {order.status === "pending" && (
                <div style={{ display: "flex", gap: 10, flexWrap: "wrap" }}>
                  <button style={buttonStyle} onClick={() => changeOrderStatus(order.id, "confirmed")}>
                    قبول الإيصال وتأكيد الطلب
                  </button>
                  <button style={{ ...buttonStyle, background: "#a52b50" }}
                    onClick={() => changeOrderStatus(order.id, "rejected")}>
                    رفض الإيصال
                  </button>
                </div>
              )}
            </article>
          ))}
        </section>

        <section style={sectionStyle}>
          <h2>إضافة منتج جديد</h2>

          <label>اسم المنتج</label>
          <input value={name} onChange={e => setName(e.target.value)}
            placeholder="اسم المنتج" style={inputStyle} />

          <label>القسم الرئيسي</label>
          <select value={category} onChange={e => {
            const next = e.target.value;
            setCategory(next);
            setSubcategory(subcategories[next][0]);
          }} style={inputStyle}>
            {Object.keys(subcategories).map(item =>
              <option key={item} value={item}>{item}</option>
            )}
          </select>

          <label>القسم الفرعي</label>
          <select value={subcategory} onChange={e => setSubcategory(e.target.value)}
            style={inputStyle}>
            {subcategories[category].map(item =>
              <option key={item} value={item}>{item}</option>
            )}
          </select>

          <label>سعر الشراء من التاجر (سري)</label>
          <input type="number" min="0" value={costPrice}
            onChange={e => setCostPrice(e.target.value)}
            placeholder="سعر الشراء" style={inputStyle} />

          <label>الكمية المتاحة في المخزون</label>
<input
  type="number"
  min="0"
  value={stock}
  onChange={e => setStock(e.target.value)}
  placeholder="الكمية المتاحة"
  style={inputStyle}
/>

<label>سعر البيع للعميل</label>
          <input type="number" min="1" value={price}
            onChange={e => setPrice(e.target.value)}
            placeholder="سعر البيع" style={inputStyle} />

          <p style={{ color: "#f28be9" }}>
            ربح القطعة: {
              costPrice && price
                ? money(Number(price) - Number(costPrice))
                : "سيظهر بعد إدخال الأسعار"
            }
          </p>

          <label>صور المنتج (حتى 5 صور)</label>
          <input type="file" accept="image/*" multiple
            onChange={e => {
              handleImages(e.target.files);
              e.target.value = "";
            }} style={inputStyle} />

          <div style={{ display: "flex", flexWrap: "wrap", gap: 10 }}>
            {images.map((src, index) => (
              <div key={index} style={{ position: "relative" }}>
                <img src={src} alt="صورة المنتج"
                  style={{ width: 85, height: 85, objectFit: "contain", background: "white", borderRadius: 10 }} />
                <button onClick={() => setImages(old => old.filter((_, i) => i !== index))}
                  style={{ position: "absolute", top: 2, right: 2, background: "#a21b55", color: "white", border: 0, borderRadius: 20 }}>
                  ×
                </button>
              </div>
            ))}
          </div>

          <button onClick={addProduct} style={buttonStyle}>+ حفظ وإضافة المنتج</button>
          {message && <p style={{ color: "#f5a3ed" }}>{message}</p>}
        </section>

        <section style={sectionStyle}>
          <h2>المنتجات ({products.length})</h2>

          {products.map(product => (
            <article key={product.id} style={orderStyle}>
              <h3>{product.name}</h3>
              <p>{product.category} — {product.subcategory}</p>
              <p>سعر الشراء: {money(product.costPrice)}</p>
              <p>سعر البيع: {money(product.price)}</p>
              <p style={{ color: "#f28be9" }}>
                ربح القطعة: {money(product.price - product.costPrice)}
              </p>

              <div style={{ display: "flex", gap: 8, flexWrap: "wrap" }}>
                {product.images.map((src, index) => (
                  <img key={index} src={src} alt={product.name}
                    style={{ width: 65, height: 65, objectFit: "contain", background: "white", borderRadius: 8 }} />
                ))}
              </div>

              <button onClick={() => {
                if (window.confirm("هل تريدين حذف المنتج؟")) {
                  saveProducts(products.filter(p => p.id !== product.id));
                }
              }} style={{ ...buttonStyle, background: "#a52b50" }}>
                حذف المنتج
              </button>
            </article>
          ))}
        </section>

      </div>
    </main>
  );
}

function Stat({ title, value }: { title: string; value: string | number }) {
  return (
    <div style={statStyle}>
      <p style={{ color: "#cbb7d8" }}>{title}</p>
      <strong style={{ color: "#f28be9", fontSize: 22 }}>{value}</strong>
    </div>
  );
}

const pageStyle: React.CSSProperties = {
  minHeight: "100vh",
  background: "#120d1b",
  color: "white",
  padding: 25,
  fontFamily: "Arial, sans-serif",
};

const sectionStyle: React.CSSProperties = {
  background: "#21172d",
  padding: 22,
  borderRadius: 16,
  margin: "25px 0",
};

const statsGrid: React.CSSProperties = {
  display: "grid",
  gridTemplateColumns: "repeat(auto-fit, minmax(160px, 1fr))",
  gap: 12,
};

const statStyle: React.CSSProperties = {
  background: "#281b35",
  padding: 18,
  borderRadius: 14,
};

const orderStyle: React.CSSProperties = {
  background: "#281b35",
  borderRadius: 14,
  padding: 18,
  marginBottom: 14,
};

const inputStyle: React.CSSProperties = {
  display: "block",
  width: "100%",
  boxSizing: "border-box",
  padding: 13,
  margin: "8px 0 18px",
  borderRadius: 9,
  border: "1px solid #654879",
  background: "#160f20",
  color: "white",
  fontSize: 16,
};

const buttonStyle: React.CSSProperties = {
  display: "block",
  width: "100%",
  padding: 14,
  marginTop: 14,
  border: 0,
  borderRadius: 10,
  background: "#b832a8",
  color: "white",
  fontSize: 16,
  fontWeight: "bold",
  cursor: "pointer",
};
